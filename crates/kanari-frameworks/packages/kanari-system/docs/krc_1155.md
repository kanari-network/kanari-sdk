# Module `0x2::krc_1155`

KRC-1155 is Kanari's semi-fungible token standard.

Each <code><a href="krc_1155.md#0x2_krc_1155_TokenType">TokenType</a></code> has an independent supply cap. Holder value is represented
by owned <code><a href="krc_1155.md#0x2_krc_1155_Balance">Balance</a></code> objects, which can be split, joined, transferred and
burned without a global address-to-balance map. This preserves the object
ownership model while supporting many indistinguishable copies per type.

<pre><code><b>public</b> <b>fun</b> <a href="krc_1155.md#0x2_krc_1155_mint">mint</a>(
    token: &<b>mut</b> <a href="krc_1155.md#0x2_krc_1155_TokenType">TokenType</a>,
    cap: &<a href="krc_1155.md#0x2_krc_1155_TokenCap">TokenCap</a>,

    ctx: &<b>mut</b> TxContext

    <b>assert</b>!(amount &gt; 0, <a href="krc_1155.md#0x2_krc_1155_E_ZERO_AMOUNT">E_ZERO_AMOUNT</a>);
    <b>let</b> token_type = <a href="object.md#0x2_object_uid_address">object::uid_address</a>(&token.id);
    <b>assert</b>!(cap.token_type == token_type, <a href="krc_1155.md#0x2_krc_1155_E_TOKEN_CAP_MISMATCH">E_TOKEN_CAP_MISMATCH</a>);
    <b>let</b> next_supply = token.total_supply + amount;
    <b>assert</b>!(next_supply &gt;= token.total_supply, <a href="krc_1155.md#0x2_krc_1155_E_SUPPLY_OVERFLOW">E_SUPPLY_OVERFLOW</a>);

    // Kanari runtime <b>requires</b> explicit save for borrowed objects
    // that are modified but not returned.
    <a href="object.md#0x2_object_save_object">object::save_object</a>(token);
    <b>let</b> <a href="balance.md#0x2_balance">balance</a> = <a href="krc_1155.md#0x2_krc_1155_Balance">Balance</a> {

        amount
    };
    // Emit <a href="krc_1155.md#0x2_krc_1155_Minted">Minted</a> <a href="event.md#0x2_event">event</a> for indexers

            token_type,

            amount
        }
    );
    <a href="balance.md#0x2_balance">balance</a>
}
