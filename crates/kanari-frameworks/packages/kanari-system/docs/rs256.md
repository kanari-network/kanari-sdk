    // 1. Signature length must match modulus (n) length (both in bytes)
    // This <b>ensures</b> the signature was generated <b>with</b> the same key size
    <b>assert</b>!(<a href="dependencies/move-stdlib/vector.md#0x1_vector_length">vector::length</a>(signature) == <a href="dependencies/move-stdlib/vector.md#0x1_vector_length">vector::length</a>(n), <a href="rs256.md#0x2_rs256_ErrorInvalidSignature">ErrorInvalidSignature</a>);

    // 2. Ensure modulus (n) length meets minimum security requirements (2048 bits = 256 bytes)
    // RSA keys smaller than 2048 bits are considered insecure
    <b>assert</b>!(<a href="dependencies/move-stdlib/vector.md#0x1_vector_length">vector::length</a>(n) &gt;= <a href="rs256.md#0x2_rs256_RSASSA_PKCS1_V1_5_MINIMUM_MODULUS_LENGTH">RSASSA_PKCS1_V1_5_MINIMUM_MODULUS_LENGTH</a> / 8, <a href="rs256.md#0x2_rs256_ErrorInvalidPubKey">ErrorInvalidPubKey</a>);

    // 3. Ensure modulus (n) length does not exceed maximum (4096 bits = 512 bytes)
    // Larger keys provide diminishing returns and impact performance
    <b>assert</b>!(<a href="dependencies/move-stdlib/vector.md#0x1_vector_length">vector::length</a>(n) &lt;= (<a href="rs256.md#0x2_rs256_RSASSA_PKCS1_V1_5_MINIMUM_MODULUS_LENGTH">RSASSA_PKCS1_V1_5_MINIMUM_MODULUS_LENGTH</a> / 8) * 2, <a href="rs256.md#0x2_rs256_ErrorInvalidPubKey">ErrorInvalidPubKey</a>);

    // 4. Ensure exponent (e) length meets minimum requirements (at least 1 byte)
    <b>assert</b>!(<a href="dependencies/move-stdlib/vector.md#0x1_vector_length">vector::length</a>(e) &gt;= <a href="rs256.md#0x2_rs256_RSASSA_PKCS1_V1_5_MINIMUM_EXPONENT_LENGTH">RSASSA_PKCS1_V1_5_MINIMUM_EXPONENT_LENGTH</a>, <a href="rs256.md#0x2_rs256_ErrorInvalidPubKey">ErrorInvalidPubKey</a>);

    // 5. Ensure exponent (e) length does not exceed maximum (512 bytes)
    <b>assert</b>!(<a href="dependencies/move-stdlib/vector.md#0x1_vector_length">vector::length</a>(e) &lt;= <a href="rs256.md#0x2_rs256_RSASSA_PKCS1_V1_5_MAXIMUM_EXPONENT_LENGTH">RSASSA_PKCS1_V1_5_MAXIMUM_EXPONENT_LENGTH</a>, <a href="rs256.md#0x2_rs256_ErrorInvalidPubKey">ErrorInvalidPubKey</a>);

    // 6. Validate exponent (e) value for security
    <b>let</b> e_len = <a href="dependencies/move-stdlib/vector.md#0x1_vector_length">vector::length</a>(e);

    // For single-byte exponents, ensure value is &gt;= 3
    // Values 1 and 2 are cryptographically weak for RSA
    <b>if</b> (e_len == 1) {
        <b>let</b> first_byte = *<a href="dependencies/move-stdlib/vector.md#0x1_vector_borrow">vector::borrow</a>(e, 0);
        <b>assert</b>!(first_byte &gt;= 3, <a href="rs256.md#0x2_rs256_ErrorInvalidPubKey">ErrorInvalidPubKey</a>);
    };

    // For multi-byte exponents, perform additional validation
    <b>if</b> (e_len &gt; 1) {
        // a) Verify exponent is not all zeros (would be invalid)
        <b>let</b> has_non_zero = <a href="rs256.md#0x2_rs256_check_exponent_not_all_zeros">check_exponent_not_all_zeros</a>(e, e_len, 0);

        // This catches malformed or intentionally obfuscated keys

        // c) Limit actual exponent size <b>to</b> prevent abnormally large values
        // After removing leading zeros, exponent should be ≤ 8 bytes (64 bits)
        // This allows common values like 65537 (0x010001) <b>while</b> rejecting extreme cases
        <b>let</b> actual_bytes = e_len - leading_zeros;
        <b>assert</b>!(actual_bytes &lt;= 8, <a href="rs256.md#0x2_rs256_ErrorInvalidPubKey">ErrorInvalidPubKey</a>);
    };
